# One source of truth for secret detection.
# Consumed by preflight.py (the commit gate), memory_redact_shim.py (the redactor in front of agent
# memory), project_audit.py and the ECC / skill-adoption supply-chain scans - so they never drift apart.
#
# Format: <id><TAB><regex><TAB><description>
# Python and .NET compatible. Every pattern is regression-tested by test_preflight.py against real leak
# shapes AND real-world strings that must not match.

supabase-mgmt	(?<![A-Za-z0-9])sbp_[A-Za-z0-9]{20,}	Supabase management/access token - can create and delete projects
supabase-svc	SUPABASE_SERVICE_ROLE[A-Z_]*\s*[:=]\s*["']?[A-Za-z0-9_\-.]{20,}	Supabase service-role key assigned a literal value
runpod	(?<![A-Za-z0-9])rpa_[A-Za-z0-9]{20,}	RunPod API key
google-api	AIza[A-Za-z0-9_\-]{35}	Google / Gemini API key
openai-like	(?<![A-Za-z0-9])sk-(?:[a-z]{2,6}-)?[A-Za-z0-9_]{20,}	OpenAI-style secret key (sk-, sk-proj-, sk-ant-, ...)
github-token	(?<![A-Za-z0-9])gh[pousr]_[A-Za-z0-9]{30,}	GitHub personal access / OAuth token
jwt	eyJ[A-Za-z0-9_\-]{10,}\.eyJ[A-Za-z0-9_\-]{10,}	JWT - includes Supabase anon and service keys
aws-akid	AKIA[0-9A-Z]{16}	AWS access key id
private-key	-----BEGIN[ A-Z]*PRIVATE KEY-----	PEM private key
stripe-like	(?<![A-Za-z0-9])sk_[A-Za-z0-9]{24,}	Stripe / RevenueCat style secret key
